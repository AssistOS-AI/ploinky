> Superseded implementation note (2026-07-09): this archival plan was updated

> config providers (`providesConfig` plus profile `configProviders`), documented

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `basic/web-publishing` Ploinky agent (custom nginx+cloudflared image, admin dashboard) plus a new ploinky "agent-published workspace vars" runtime feature, so public-exposure env vars (`ONLYOFFICE_PUBLIC_URL`, `WEB_PUBLISHING_CLOUDFLARED_TUNNEL_TOKEN`, `WEBMEET_*`) are generated/persisted/published by the agent instead of being passed into the Ploinky box container.

**Architecture:** One agent container runs nginx + cloudflared + the standard AgentServer under a supervisor (the `basic/cloudflared` dual-command pattern). Its admin-tagged MCP tools (driven by an Explorer settings plugin) write an agent-local config store, regenerate nginx/cloudflared state, and push derived consumer values through a new router-owned, Agent-Assertion-authenticated endpoint `POST /api/router/startup-config-providers` into an encrypted store `.ploinky/data/startup-config-providers.enc`. Manifest env resolution consults that store after operator sources and before manifest defaults, so consumers pick published values up via the existing envhash-driven container recreate on the next start/restart.

**Tech Stack:** Node.js ESM (Node 20+), `node:test` + `node:assert/strict`, bash lifecycle hooks, nginx, cloudflared, Cloudflare REST API v4, Docker/Podman multi-stage images, GitHub Actions image publishing.

**Design source:** `ploinky/docs/superpowers/specs/2026-07-09-web-publishing-agent-design.md` (answers the 12 design questions; read it before executing).

`ploinky/cli/services/startupConfigProviders.js` (match masterKey.js's module system; shown as ESM — convert to CJS `require/module.exports` if masterKey.js is CJS, mirroring its own imports):

```js
import crypto from 'node:crypto';
import fs from 'node:fs';
import path from 'node:path';
import { deriveSubkey } from './masterKey.js';

// Agent-published workspace variables (design 2026-07-09-web-publishing-agent-design.md).
// Encrypted with a dedicated HKDF purpose so a leak of this store's key cannot
// decrypt .secrets or passwords (DS011: new persistent secret => new purpose label).
const PURPOSE = 'storage/startup-config-providers';
const ALGORITHM = 'aes-256-gcm';
const IV_BYTES = 12;
const TAG_BYTES = 16;
const NAME_RE = /^[A-Z][A-Z0-9_]{0,127}$/;
const RESERVED_EXACT = new Set(['WEBDASHBOARD_TOKEN', 'APP_NAME']);

function workspaceRoot() {
    return process.env.PLOINKY_WORKSPACE_ROOT || process.cwd();
}

export const PUBLISHED_VARS_FILE = path.join(workspaceRoot(), '.ploinky', 'data', 'startup-config-providers.enc');

function storeFile() {
    // Recomputed per call so tests can point PLOINKY_WORKSPACE_ROOT at a temp dir.
    return path.join(workspaceRoot(), '.ploinky', 'data', 'startup-config-providers.enc');
}

export function isValidPublishedName(name) {
    return typeof name === 'string' && NAME_RE.test(name);
}

export function isReservedPublishedName(name) {
    const n = String(name || '');
    return n.startsWith('PLOINKY_') || RESERVED_EXACT.has(n);
}

function decrypt(packed) {
    const buf = Buffer.from(String(packed), 'base64');
    const iv = buf.subarray(0, IV_BYTES);
    const tag = buf.subarray(IV_BYTES, IV_BYTES + TAG_BYTES);
    const ciphertext = buf.subarray(IV_BYTES + TAG_BYTES);
    const decipher = crypto.createDecipheriv(ALGORITHM, deriveSubkey(PURPOSE), iv);
    decipher.setAuthTag(tag);
    const plain = Buffer.concat([decipher.update(ciphertext), decipher.final()]);
    return JSON.parse(plain.toString('utf8'));
}

function encrypt(doc) {
    const iv = crypto.randomBytes(IV_BYTES);
    const cipher = crypto.createCipheriv(ALGORITHM, deriveSubkey(PURPOSE), iv);
    const plaintext = Buffer.from(JSON.stringify(doc), 'utf8');
    const ciphertext = Buffer.concat([cipher.update(plaintext), cipher.final()]);
    return Buffer.concat([iv, cipher.getAuthTag(), ciphertext]).toString('base64');
}

export function readPublishedVars() {
    const file = storeFile();
    let raw;
    try {
        raw = fs.readFileSync(file, 'utf8');
    } catch {
        return {};
    }
    try {
        const doc = decrypt(raw.trim());
        if (!doc || doc.version !== 1 || typeof doc.vars !== 'object' || !doc.vars) return {};
        return doc.vars;
    } catch {
        // Corrupt or wrong-key store: treat as absent so env resolution falls
        // back to operator sources and manifest defaults. Recovery: delete file.
        return {};
    }
}

export function getPublishedValue(name) {
    const entry = readPublishedVars()[name];
    return entry && typeof entry.value === 'string' ? entry.value : undefined;
}

export function setPublishedVars(entries, publisher) {
    const vars = readPublishedVars();
    const written = [];
    const deleted = [];
    const updatedAt = new Date().toISOString();
    for (const [name, value] of Object.entries(entries || {})) {
        if (!isValidPublishedName(name) || isReservedPublishedName(name)) {
            throw new Error(`published var name not allowed: ${name}`);
        }
        if (value === null) {
            if (name in vars) { delete vars[name]; deleted.push(name); }
            continue;
        }
        if (typeof value !== 'string' || value.length > 8192) {
            throw new Error(`published var value invalid for ${name}`);
        }
        vars[name] = { value, publisher: String(publisher || ''), updatedAt };
        written.push(name);
    }
    const file = storeFile();
    fs.mkdirSync(path.dirname(file), { recursive: true });
    const tmp = `${file}.tmp-${process.pid}`;
    fs.writeFileSync(tmp, `${encrypt({ version: 1, vars })}\n`, { mode: 0o600 });
    fs.renameSync(tmp, file);
    return { written, deleted };
}

export default { readPublishedVars, getPublishedValue, setPublishedVars, isValidPublishedName, isReservedPublishedName, PUBLISHED_VARS_FILE };
```

Note: if `PUBLISHED_VARS_FILE` as a static export conflicts with the per-call `storeFile()` (tests chdir before import), export a `getPublishedVarsFile()` function instead and update the test — keep one source of truth.

```js
import { readEnabledAgentManifest, loadRoutingConfig } from './httpServiceRoutes.js';
import { deriveAgentPrincipalId } from '../services/agentIdentity.js';
import { verifyAgentAssertion } from './mcp-proxy/invocationMinter.js';
import { createTokenReplayCache } from './security/tokens/JwsCodec.js';
import { computeRchHttp, sha256RawBodyHash } from '../../Agent/lib/requestHash.mjs';
import { setPublishedVars, isValidPublishedName, isReservedPublishedName } from '../services/startupConfigProviders.js';

/**
 * startupConfigProvidersRoute.js — router-owned, agent-authenticated write surface for
 * agent-published workspace variables (design 2026-07-09).
 *
 * POST /api/router/startup-config-providers with an HTTP Agent Assertion
 * (tool `__published_vars__`, rch over the exact body bytes). The router derives
 * the caller identity from the verified assertion, loads THAT agent's enabled
 * manifest, and accepts only names listed in its `publishesVars` allowlist.
 * Reserved names (PLOINKY_*, WEBDASHBOARD_TOKEN, APP_NAME) are always denied.
 * The batch is atomic: one bad name rejects the whole request.
 * Dispatched BEFORE the session-auth gate, like /api/router/openai-agent-discovery.
 */

export const PUBLISHED_VARS_PATH = '/api/router/startup-config-providers';
export const PUBLISHED_VARS_TOOL = '__published_vars__';
const MAX_BODY_BYTES = 64 * 1024;
const MAX_VARS_PER_CALL = 64;

const publishReplayCache = createTokenReplayCache({ maxSize: 4096 });

function readAuthorizationBearer(req) {
    const raw = req?.headers?.authorization ?? req?.headers?.Authorization;
    const value = Array.isArray(raw) ? raw[0] : raw;
    if (typeof value !== 'string') return '';
    const trimmed = value.trim();
    return /^bearer\s+/i.test(trimmed) ? trimmed.replace(/^bearer\s+/i, '').trim() : '';
}

function sendJson(res, statusCode, body) {
    const data = Buffer.from(JSON.stringify(body));
    res.writeHead(statusCode, { 'Content-Type': 'application/json', 'Content-Length': data.length });
    res.end(data);
}

function findManifestForAgentId(agentId, routing, readManifest) {
    const routes = routing?.routes && typeof routing.routes === 'object' ? routing.routes : {};
    for (const [routeKey, route] of Object.entries(routes)) {
        if (!route || route.disabled) continue;
        const repo = String(route.repo || '').trim();
        const agent = String(route.agent || '').trim();
        if (!repo || !agent) continue;
        let principal = null;
        try { principal = deriveAgentPrincipalId(repo, agent); } catch { continue; }
        if (principal === agentId) {
            try { return readManifest(routeKey, routes) || null; } catch { return null; }
        }
    }
    return null;
}

function collectBody(req, limit) {
    return new Promise((resolve, reject) => {
        const chunks = [];
        let size = 0;
        req.on('data', (chunk) => {
            size += chunk.length;
            if (size > limit) { reject(new Error('body_too_large')); req.destroy?.(); return; }
            chunks.push(chunk);
        });
        req.on('end', () => resolve(Buffer.concat(chunks)));
        req.on('error', reject);
    });
}

export function handlePublishedVarsRoute(req, res, parsedUrl, {
    routing = loadRoutingConfig(),
    readManifest = readEnabledAgentManifest,
    replayCache = publishReplayCache,
    hasOperatorValue = defaultHasOperatorValue,
} = {}) {
    const pathname = parsedUrl?.pathname || '';
    if (pathname !== PUBLISHED_VARS_PATH) return false;

    const method = String(req?.method || '').toUpperCase();
    if (method !== 'POST') {
        res.writeHead(405, { 'Content-Type': 'application/json', Allow: 'POST' });
        res.end(JSON.stringify({ ok: false, error: 'method_not_allowed' }));
        return true;
    }

    (async () => {
        let body;
        try {
            body = await collectBody(req, MAX_BODY_BYTES);
        } catch {
            sendJson(res, 413, { ok: false, error: 'body_too_large' });
            return;
        }

        const token = readAuthorizationBearer(req);
        const rch = computeRchHttp({ method: 'POST', path: PUBLISHED_VARS_PATH, query: '', bodyHash: sha256RawBodyHash(body) });
        let verified;
        try {
            verified = verifyAgentAssertion({
                token, method: 'POST', path: PUBLISHED_VARS_PATH,
                tool: PUBLISHED_VARS_TOOL, rch, targetAgentId: 'ploinky-router', replayCache,
            });
        } catch {
            sendJson(res, 401, { ok: false, error: 'agent_assertion_required' });
            return;
        }
        const sourceAgentId = String(verified?.iss || verified?.payload?.iss || '').trim();
        if (!/^agent:[^/]+\/.+$/.test(sourceAgentId)) {
            sendJson(res, 401, { ok: false, error: 'agent_assertion_required' });
            return;
        }

        let payload;
        try { payload = JSON.parse(body.toString('utf8')); } catch { payload = null; }
        const vars = payload && typeof payload.vars === 'object' && payload.vars ? payload.vars : null;
        const names = vars ? Object.keys(vars) : [];
        if (!vars || names.length === 0 || names.length > MAX_VARS_PER_CALL) {
            sendJson(res, 400, { ok: false, error: 'invalid_payload' });
            return;
        }

        const manifest = findManifestForAgentId(sourceAgentId, routing, readManifest);
        const allow = new Set(Array.isArray(manifest?.publishesVars) ? manifest.publishesVars.map(String) : []);
        const refused = names.filter((n) => !isValidPublishedName(n) || isReservedPublishedName(n) || !allow.has(n));
        if (refused.length) {
            console.log(`[startup-config-providers] refused ${sourceAgentId}: ${refused.join(', ')}`);
            sendJson(res, 403, { ok: false, error: 'var_not_publishable', names: refused });
            return;
        }

        let result;
        try {
            result = setPublishedVars(vars, sourceAgentId);
        } catch {
            sendJson(res, 500, { ok: false, error: 'publish_failed' });
            return;
        }
        const shadowed = names.filter((n) => { try { return hasOperatorValue(n); } catch { return false; } });
        console.log(`[startup-config-providers] ${sourceAgentId} wrote [${result.written.join(', ')}] deleted [${result.deleted.join(', ')}]`);
        sendJson(res, 200, { ok: true, written: result.written, deleted: result.deleted, shadowed });
    })();

    return true;
}

function defaultHasOperatorValue(name) {
    if (Object.prototype.hasOwnProperty.call(process.env, name)) return true;
    // Checks the operator var store without importing the deny-listed filename here:
    // resolution precedence is what matters; reuse the vars listing service.
    try {
        // Lazy import to avoid a cycle: envVarCommands' service exposes the names.
        const { listOperatorVarNames } = requireOperatorVarNames();
        return listOperatorVarNames().includes(name);
    } catch {
        return false;
    }
}

let cachedNamesModule = null;
function requireOperatorVarNames() {
    if (!cachedNamesModule) {
        throw new Error('operator var name listing not wired');
    }
    return cachedNamesModule;
}
export function wireOperatorVarNames(mod) { cachedNamesModule = mod; }

export default { PUBLISHED_VARS_PATH, PUBLISHED_VARS_TOOL, handlePublishedVarsRoute };
```

```js
import { signAgentHttpAssertion } from './agentAssertion.mjs';

/**
 * startupConfigProvidersClient.mjs — agent-side helper for the router's
 * POST /api/router/startup-config-providers surface (agent-published workspace vars).
 * The caller must be an enabled agent whose manifest declares the names in
 * `publishesVars`. Values are strings; null deletes. Never log values.
 */

export const PUBLISHED_VARS_PATH = '/api/router/startup-config-providers';
export const PUBLISHED_VARS_TOOL = '__published_vars__';

export async function publishWorkspaceVars(vars, {
    routerUrl = process.env.PLOINKY_ROUTER_URL || 'http://host.containers.internal:8080',
    env = process.env,
    fetchImpl = fetch,
} = {}) {
    const body = Buffer.from(JSON.stringify({ vars }));
    const token = signAgentHttpAssertion({
        method: 'POST', path: PUBLISHED_VARS_PATH, query: '', body,
        targetAgent: 'ploinky-router', tool: PUBLISHED_VARS_TOOL, env,
    });
    const res = await fetchImpl(new URL(PUBLISHED_VARS_PATH, routerUrl), {
        method: 'POST',
        headers: { 'content-type': 'application/json', authorization: `Bearer ${token}` },
        body,
    });
    let parsed = null;
    try { parsed = await res.json(); } catch { parsed = null; }
    if (!res.ok || !parsed?.ok) {
        const code = parsed?.error || `http_${res.status}`;
        throw new Error(`publishWorkspaceVars failed: ${code}`);
    }
    return parsed;
}

export default { publishWorkspaceVars, PUBLISHED_VARS_PATH, PUBLISHED_VARS_TOOL };
```

```js
const CONNECTION_UPGRADE_MAP = `map $http_upgrade $connection_upgrade {
    default upgrade;
    '' close;
}`;

function serverBlock({ hostname, origin, websocket }, httpPort) {
    const ws = websocket ? `
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection $connection_upgrade;` : '';
    return `server {
    listen ${httpPort};
    server_name ${hostname};
    client_max_body_size 100m;
    location / {
        proxy_pass ${origin};
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_read_timeout 300s;${ws}
    }
}`;
}

export function buildNginxConf({ httpPort = 8081, exposures = [] } = {}) {
    const port = Number(httpPort);
    if (!Number.isInteger(port) || port < 1 || port > 65535) {
        throw new Error(`invalid nginx http port: ${httpPort}`);
    }
    const blocks = exposures.map((e) => serverBlock(e, port)).join('\n\n');
    return `worker_processes auto;
pid /tmp/web-publishing-nginx.pid;
error_log /dev/stderr warn;
events { worker_connections 1024; }
http {
    access_log /dev/stdout;
    ${CONNECTION_UPGRADE_MAP}

${blocks}

    server {
        listen ${port} default_server;
        return 444;
    }
}
`;
}

export default { buildNginxConf };
```
